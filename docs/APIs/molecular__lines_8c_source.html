228 #endif

262  }

264  snprintf(c.o3_ctm_dir,
265  DIR_PATH_LEN,

267  o3_ctm_dir);
268  c.o3_cc = NULL;
269  }
270  else
271  {
272  c.use_o3_ctm = 0;
273  }

275  /*Reserve memory.*/
276  c.num_molecules = 0;
277  check(malloc_ptr((void **)(&(c.line_params)),
278  sizeof(*(c.line_params))*MAX_NUM_MOLECULES));
279  not_null(c.line_params);

281  {

283  size_t num_elements = c.num_levels;

383  HANDLE_ERROR(cudaFree(c->Ns));
384  HANDLE_ERROR(cudaFree(c->Psavg));
385  HANDLE_ERROR(cudaFree(c->gamma));
386  HANDLE_ERROR(cudaFree(c->Pshift));
387  HANDLE_ERROR(cudaFree(c->s));
388  HANDLE_ERROR(cudaFree(c->tau));
389  check(free_line_params_device(&(c->lines)));
390 #endif
391  }
392  else
393  {
394  free(c->x);
395  free(c->Pavg);
396  free(c->Tavg);
397  free(c->N);
398  free(c->Ns);
399  free(c->Psavg);
400  free(c->snn_ref);
401  free(c->gamma);
402  free(c->Pshift);
403  free(c->s);
404  }
405  if (c->use_h2o_ctm && c->h2o_cc != NULL)
406  {
407  if (c->gpu_id != HOST_ONLY)
408  {
409  check(remove_water_vapor_coefs_from_device(c->h2o_cc));

411  else
412  {
413  check(free_water_vapor_continuum_coefs(c->h2o_cc));
414  }
415  free(c->h2o_cc);
416  }
417  if (c->use_o3_ctm && c->o3_cc != NULL)
418  {
419  if (c->gpu_id != HOST_ONLY)
420  {
421  check(remove_ozone_coefs_from_device(c->o3_cc));
422  }
423  else
424  {
425  check(free_ozone_continuum_coefs(c->o3_cc));
426  }
427  free(c->o3_cc);
428  }
429  free(c);
430  *context = NULL;
431  return SUCCESS;
432 }

439 int grt_add_molecule(GrtContext_t *context,

444 {
445  not_null(context);
446  not_null(hitran_filepath);
447  int index = (context->num_molecules)++;
448  in_range(context->num_molecules,1,MAX_NUM_MOLECULES);
449  LineFlags_t flags = {((unsigned int) -1),1,0};
450  double w0;
451  if (min_line_center_wavenumber != NULL)
452  {
453  in_range(*min_line_center_wavenumber,MIN_WAVENUMBER,MAX_WAVENUMBER);
454  w0 = *min_line_center_wavenumber;
455  }
456  else
457  {
458  w0 = context->w0;
459  }
460  double wn;
461  if (max_line_center_wavenumber != NULL)
462  {
463  in_range(*max_line_center_wavenumber,MIN_WAVENUMBER,MAX_WAVENUMBER);
464  wn = *max_line_center_wavenumber;
465  }
466  else
467  {
468  wn = context->wn;
469  }
470  min_check(wn,w0);
471  check(parse_hitran_file(&(context->line_params[index]),
472  hitran_filepath,
473  flags,
474  w0,
475  wn));
476  char mol_name[8];
477  int m = context->line_params[index]->mol;
478  check(get_mol_name(m,
479  mol_name,
480  8));
481  log_mesg("Using %s (%u lines in range %e - %e [1/cm]).",
482  mol_name,
483  context->line_params[index]->num_lines,
484  w0,
485  wn);
486  if (m == H2O && context->use_h2o_ctm)
487  {
488  log_mesg("Using the %s continuum.",
489  mol_name);
490 
491  /*Read in the water vapor continuum coefficients.*/
492  WaterVaporContinuumCoefs_t h2o_cc;
493  check(get_water_vapor_continuum_coefs(&h2o_cc,
494  context->h2o_ctm_dir,
495  context->num_wpoints,
496  context->w0,
497  context->wres));
498  check(malloc_ptr((void **) (&context->h2o_cc),
499  sizeof(*(context->h2o_cc))));
500  if (context->gpu_id != HOST_ONLY)
501  {
502 #ifdef __NVCC__
503  HANDLE_ERROR(cudaSetDevice(context->gpu_id));
504 #endif
505  check(put_water_vapor_coefs_on_device(&h2o_cc,
506  context->h2o_cc));
507  }
508  else
509  {
510  memcpy(context->h2o_cc,
511  &h2o_cc,
512  sizeof(*(context->h2o_cc)));
513  }
514  }
515 
516  if (m == O3 && context->use_o3_ctm)
517  {
518  log_mesg("Using the %s continuum.",
519  mol_name);
520 
521  /*Read in the ozone continuum coefficients.*/
522  OzoneContinuumCoefs_t o3_cc;
523  check(get_ozone_continuum_coefs(&o3_cc,
524  context->o3_ctm_dir,
525  context->num_wpoints,
526  context->w0,
527  context->wres));
528  check(malloc_ptr((void **) (&context->o3_cc),
529  sizeof(*(context->o3_cc))));
530  if (context->gpu_id != HOST_ONLY)
531  {
532 #ifdef __NVCC__
533  HANDLE_ERROR(cudaSetDevice(context->gpu_id));
534 #endif
535  check(put_ozone_coefs_on_device(&o3_cc,
536  context->o3_cc));
537  }
538  else
539  {
540  memcpy(context->o3_cc,
541  &o3_cc,
542  sizeof(*(context->o3_cc)));
543  }
544  }
545 
546  not_null(molecule_id);
547  *molecule_id = index;
548  return SUCCESS;
549 }

596  not_null(pressure);

605  context->gpu_id);
606 #ifdef __NVCC__

609  HANDLE_ERROR(cudaMemcpy(context->P,
610  pressure,

612  cudaMemcpyHostToDevice));
613  HANDLE_ERROR(cudaMemcpy(context->T,
614  temperature,
615  sizeof(*temperature)*num_elements,
616  cudaMemcpyHostToDevice));
617  check(launch(context->num_levels,
618  context->P,
619  context->T,
620  context->x,
621  context->Pavg,
622  context->Tavg,
623  context->N,
624  context->Ns,
625  context->Psavg,
626  context->gamma,
627  context->Pshift,
628  context->s,
629  context->lines,
630  context->num_molecules,
631  context->line_params,
632  context->w0,
633  context->wres,
634  context->num_wpoints,
635  context->wcutoff,
636  context->use_h2o_ctm,
637  context->h2o_cc,
638  context->use_o3_ctm,
639  context->o3_cc,
640  context->tau));
641  num_elements = context->num_layers*context->num_wpoints;
642  HANDLE_ERROR(cudaMemcpy(optical_depth,
643  context->tau,
644  sizeof(*optical_depth)*num_elements,
645  cudaMemcpyDeviceToHost));
646 #endif
647  }
648  else
649  {
650  log_mesg("Calculating optical depths for %d molecules in %d"
651  " atmospheric layers on the host CPU.",
652  context->num_molecules,
653  context->num_levels-1);
654  check(launch_h(context->num_levels,
655  pressure,
656  temperature,
657  context->x,
658  context->Pavg,
659  context->Tavg,
660  context->N,
661  context->Ns,
662  context->Psavg,
663  context->snn_ref,
664  context->gamma,
665  context->Pshift,
